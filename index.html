  <script src="https://unpkg.com/react@17/umd/react.production.min.js" crossorigin></script>

  <script src="https://unpkg.com/react-dom@17/umd/react-dom.production.min.js" crossorigin></script>

    body { font-family: 'Sarabun', sans-serif; background: linear-gradient(135deg, #f0f9ff 0%, #e0e7ff 50%, #fae8ff 100%); color: #334155; min-height: 100vh; }

    .glass-card { background: rgba(255, 255, 255, 0.9); backdrop-filter: blur(12px); border: 1px solid rgba(255, 255, 255, 1); box-shadow: 0 10px 40px -10px rgba(0, 0, 0, 0.08); }

    .period-chip { transition: all 0.2s ease; background: white; border: 1px solid #e2e8f0; }

    .period-chip:hover { border-color: #3b82f6; transform: translateY(-2px); }

    .period-chip.active { background: linear-gradient(135deg, #3b82f6, #8b5cf6); color: white; border-color: transparent; box-shadow: 0 4px 12px rgba(139, 92, 246, 0.3); }

    .gradient-text { background: linear-gradient(135deg, #2563eb, #9333ea); -webkit-background-clip: text; -webkit-text-fill-color: transparent; }

  <div id="root"></div>


  <script type="text/babel">

    const { useState, useEffect } = React;


    const DAYS_TH = ['อาทิตย์', 'จันทร์', 'อังคาร', 'พุธ', 'พฤหัสบดี', 'ศุกร์', 'เสาร์'];

    const MONTHS_TH_SHORT = ['ม.ค.', 'ก.พ.', 'มี.ค.', 'เม.ย.', 'พ.ค.', 'มิ.ย.', 'ก.ค.', 'ส.ค.', 'ก.ย.', 'ต.ค.', 'พ.ย.', 'ธ.ค.'];

    const PERIODS = [1, 2, 3, 4, 5, 6, 7, 8];

    const LEAVE_TYPES = ['ลากิจ', 'ลาป่วย', 'ไปราชการ', 'ประชุม'];

    

    // ตั้งค่าบทบาทพิเศษ

    const SPECIAL_ROLES = {

      'นางกรกนก หาญไชยนะ': 'หัวหน้ากลุ่มบริหาร',

      'นายพัลลภ ทองหล่อ': 'หัวหน้ากลุ่มบริหาร',

      'นายณรงค์ศักดิ์ หงษ์ตะนุ': 'หัวหน้ากลุ่มบริหาร',

      'นางปุณญิสา ศรีชมภู': 'เจ้าหน้าที่สำนักงาน'

    };


    function App() {

      const [activeTab, setActiveTab] = useState('manage'); 

      const [isLoading, setIsLoading] = useState(true);

      const [isSaving, setIsSaving] = useState(false);

      const [hasSaved, setHasSaved] = useState(false);

      

      const [showConfirmModal, setShowConfirmModal] = useState(false);

      const [showSuccessModal, setShowSuccessModal] = useState(false);


      const [scheduleData, setScheduleData] = useState([]);

      const [teachersList, setTeachersList] = useState([]);

      const [stats, setStats] = useState({});

      const [leaveRequests, setLeaveRequests] = useState([]);

      const [assignments, setAssignments] = useState([]);

      

      const [selectedTeacherLeave, setSelectedTeacherLeave] = useState(null);

      const [selectedScheduleDay, setSelectedScheduleDay] = useState('จันทร์'); 

      

      const [currentForm, setCurrentForm] = useState({

        teacherName: '', reason: 'ลากิจ', date: new Date().toISOString().split('T')[0], selectedPeriods: [] 

      });


      useEffect(() => {

        google.script.run.withSuccessHandler((response) => {

          const res = JSON.parse(response);

          if (res.success) {

            setScheduleData(res.data);

            const unique = {};

            res.data.forEach(item => { if (item.teacherName && !unique[item.teacherName]) unique[item.teacherName] = { name: item.teacherName, dept: item.dept }; });

            const tList = Object.values(unique);

            setTeachersList(tList);

            

            google.script.run.withSuccessHandler((s) => { 

              let parsedStats = JSON.parse(s);

              let initialStats = {};

              tList.forEach(t => { 

                initialStats[t.name] = parsedStats[t.name] || { leaveCount: 0, subCount: 0, leaveHistory: [] }; 

                if(!initialStats[t.name].leaveHistory) initialStats[t.name].leaveHistory = [];

              });

              setStats(initialStats); 

              setIsLoading(false); 

            }).getStats();

          }

        }).getScheduleData();

      }, []);


      const formatThaiDate = (dateStr) => {

        const d = new Date(dateStr);

        return `${DAYS_TH[d.getDay()]} ${d.getDate()} ${MONTHS_TH_SHORT[d.getMonth()]} ${((d.getFullYear() + 543).toString().slice(-2))}`;

      };


      const getDayNameFromDate = (dateStr) => DAYS_TH[new Date(dateStr).getDay()];


      const availableClasses = currentForm.teacherName ? 

        scheduleData.filter(s => s.teacherName === currentForm.teacherName && s.day === getDayNameFromDate(currentForm.date) && s.subject !== 'ว่าง') : [];


      const togglePeriod = (p) => {

        const pSet = new Set(currentForm.selectedPeriods);

        pSet.has(p) ? pSet.delete(p) : pSet.add(p);

        setCurrentForm({...currentForm, selectedPeriods: Array.from(pSet).sort()});

      };


      const handleAddLeave = () => {

        if (!currentForm.teacherName || currentForm.selectedPeriods.length === 0) return alert('กรุณาเลือกครูและคลิกเลือกคาบสอนที่ต้องการจัดแทน');

        setLeaveRequests([...leaveRequests, { ...currentForm, id: Date.now(), displayDate: formatThaiDate(currentForm.date) }]);

        setCurrentForm({...currentForm, teacherName: '', selectedPeriods: []});

      };


      const calculateSubstitutes = () => {

        let newAss = [];

        let dailyLimit = {};

        

        leaveRequests.forEach(leave => {

          const dayName = getDayNameFromDate(leave.date);

          const classesToCover = scheduleData.filter(s => s.teacherName === leave.teacherName && s.day === dayName && leave.selectedPeriods.includes(s.period));


          classesToCover.forEach(cls => {

            let freeCandidates = scheduleData.filter(s => s.day === dayName && s.period === cls.period && s.subject === 'ว่าง');

            freeCandidates = freeCandidates.filter(c => !leaveRequests.some(l => l.teacherName === c.teacherName && l.date === leave.date && l.selectedPeriods.includes(c.period)));

            

            let scoredFree = freeCandidates.map(c => {

              let score = 100;

              const tInfo = teachersList.find(t => t.name === c.teacherName);

              if (tInfo?.dept === cls.dept) score += 30;

              if (dailyLimit[`${leave.date}_${c.teacherName}`]) score -= 60;

              score -= (stats[c.teacherName]?.subCount || 0) * 5;

              if (SPECIAL_ROLES[c.teacherName]) score -= 500; 

              return { name: c.teacherName, score, dept: tInfo?.dept, isFree: true };

            }).sort((a,b) => b.score - a.score);


            let busyCandidates = teachersList.filter(t => !freeCandidates.some(fc => fc.teacherName === t.name));

            busyCandidates = busyCandidates.filter(c => !leaveRequests.some(l => l.teacherName === c.name && l.date === leave.date && l.selectedPeriods.includes(cls.period)));

            

            let scoredBusy = busyCandidates.map(c => {

               return { name: c.name, score: -1000, dept: c.dept, isFree: false };

            });


            const allOptions = [...scoredFree, ...scoredBusy];

            const selected = scoredFree.length > 0 ? scoredFree[0].name : 'ไม่มีครูว่าง';

            if (selected !== 'ไม่มีครูว่าง') dailyLimit[`${leave.date}_${selected}`] = true;


            newAss.push({

              id: Math.random(), leaveId: leave.id, leaveName: cls.teacherName, reason: leave.reason,

              date: leave.date, displayDate: leave.displayDate,

              period: cls.period, subject: cls.subject, grade: cls.grade, room: cls.room,

              subName: selected, options: allOptions

            });

          });

        });

        newAss.sort((a, b) => a.period - b.period);

        setAssignments(newAss);

      };


      const handleCopyText = (leaveReq, reqAssignments) => {

        let text = `วันนี้ ครู${leaveReq.teacherName} ลา${leaveReq.reason} ขออนุญาตจัดสอนแทนดังนี้\n`;

        reqAssignments.forEach(a => {

            text += `คาบที่ ${a.period} (${a.subject} ${a.grade} ห้อง ${a.room}) -> สอนแทนโดย: ${a.subName}\n`;

        });

        navigator.clipboard.writeText(text).then(() => {

            alert('คัดลอกข้อความสำเร็จแล้ว สามารถนำไปวางได้เลยครับ!');

        });

      };


      const handleDownloadImage = (dateStr) => {

        const elementId = `master-grid-${dateStr}`;

        const element = document.getElementById(elementId);

        if (element) {

          html2canvas(element, { scale: 2, backgroundColor: '#ffffff' }).then(canvas => {

            const link = document.createElement('a');

            link.download = `ตารางสรุปสอนแทน_${dateStr}.png`;

            link.href = canvas.toDataURL('image/png');

            link.click();

          });

        }

      };


      const executeSave = () => {

        setIsSaving(true);

        let newStats = { ...stats };

        

        leaveRequests.forEach(req => { 

          if(newStats[req.teacherName]) {

            newStats[req.teacherName].leaveCount += 1;

            newStats[req.teacherName].leaveHistory.push({

              date: req.date, displayDate: req.displayDate, reason: req.reason, periods: req.selectedPeriods.join(', ')

            });

          } 

        });


        assignments.forEach(ass => {

          if (ass.subName !== 'ไม่มีครูว่าง' && newStats[ass.subName]) {

            newStats[ass.subName].subCount += 1;

          }

        });

        

        setStats(newStats);

        google.script.run.saveStats(JSON.stringify(newStats));

        google.script.run.withSuccessHandler(() => {

          setIsSaving(false);

          setShowConfirmModal(false);

          setHasSaved(true);

          setShowSuccessModal(true); 

        }).saveToSheet(leaveRequests, assignments);

      };


      const handleClearAll = () => {

         setLeaveRequests([]);

         setAssignments([]);

         setHasSaved(false);

      };


      if (isLoading) return <div className="h-screen flex flex-col items-center justify-center text-indigo-600 font-bold"><div className="animate-spin rounded-full h-12 w-12 border-b-4 border-indigo-600 mb-4"></div>กำลังเชื่อมต่อฐานข้อมูล Sheet...</div>;


      const datesInAssignments = [...new Set(assignments.map(a => a.date))];


      return (

        <div className="max-w-7xl mx-auto p-4 md:p-8">

          

          <header className="mb-8 flex flex-col md:flex-row justify-between items-center bg-white p-4 rounded-3xl shadow-sm border border-slate-100">

            <div className="text-center md:text-left mb-4 md:mb-0 ml-4">

              <h1 className="text-2xl md:text-3xl font-bold gradient-text">ระบบจัดสอนแทน โรงเรียนพระนารายณ์</h1>

            </div>

            <div className="flex bg-slate-100 p-1 rounded-2xl overflow-x-auto">

              <button onClick={() => setActiveTab('manage')} className={`px-4 py-2.5 rounded-xl font-bold text-sm whitespace-nowrap transition-all ${activeTab === 'manage' ? 'bg-white text-indigo-700 shadow-sm' : 'text-slate-500 hover:text-slate-700'}`}>📅 จัดตารางสอนแทน</button>

              <button onClick={() => setActiveTab('schedule')} className={`px-4 py-2.5 rounded-xl font-bold text-sm whitespace-nowrap transition-all ${activeTab === 'schedule' ? 'bg-white text-indigo-700 shadow-sm' : 'text-slate-500 hover:text-slate-700'}`}>🗓️ ตารางสอนรวม</button>

              <button onClick={() => setActiveTab('dashboard')} className={`px-4 py-2.5 rounded-xl font-bold text-sm whitespace-nowrap transition-all ${activeTab === 'dashboard' ? 'bg-white text-indigo-700 shadow-sm' : 'text-slate-500 hover:text-slate-700'}`}>📊 แดชบอร์ดสถิติ</button>

            </div>

          </header>


          {activeTab === 'manage' && (

            <div className="grid grid-cols-1 lg:grid-cols-12 gap-8">

              <div className="lg:col-span-4 space-y-6">

                <div className="glass-card rounded-3xl p-6">

                  <h2 className="text-xl font-bold mb-5 flex items-center text-indigo-800"><span className="w-2 h-6 bg-indigo-500 rounded-full mr-2"></span>บันทึกการลา</h2>

                  <div className="space-y-4">

                    <div>

                      <label className="text-xs text-indigo-600 font-bold uppercase tracking-widest mb-1 block">เลือกครูผู้ลา</label>

                      <select className="w-full bg-white border border-slate-200 rounded-xl p-3 text-sm focus:ring-2 focus:ring-indigo-400 outline-none shadow-sm" value={currentForm.teacherName} onChange={e => setCurrentForm({...currentForm, teacherName: e.target.value, selectedPeriods: []})}>

                        <option value="">-- เลือกรายชื่อ --</option>

                        {teachersList.map(t => <option key={t.name} value={t.name}>{t.name}</option>)}

                      </select>

                    </div>


                    <div className="grid grid-cols-2 gap-4">

                      <div>

                        <label className="text-xs text-indigo-600 font-bold uppercase tracking-widest mb-1 block">ประเภท</label>

                        <select className="w-full bg-white border border-slate-200 rounded-xl p-3 text-sm focus:ring-2 focus:ring-indigo-400 outline-none shadow-sm" value={currentForm.reason} onChange={e => setCurrentForm({...currentForm, reason: e.target.value})}>

                          {LEAVE_TYPES.map(r => <option key={r} value={r}>{r}</option>)}

                        </select>

                      </div>

                      <div>

                        <label className="text-xs text-indigo-600 font-bold uppercase tracking-widest mb-1 block">วันที่ลา</label>

                        <input type="date" className="w-full bg-white border border-slate-200 rounded-xl p-3 text-sm focus:ring-2 focus:ring-indigo-400 outline-none shadow-sm" value={currentForm.date} onChange={e => setCurrentForm({...currentForm, date: e.target.value, selectedPeriods: []})} />

                      </div>

                    </div>


                    {currentForm.teacherName && (

                      <div className="bg-slate-50 p-3 rounded-2xl border border-slate-100">

                        <label className="text-xs text-indigo-600 font-bold uppercase tracking-widest mb-2 block">คลิกเลือกคาบที่ต้องการจัดสอนแทน</label>

                        <div className="grid grid-cols-4 gap-2">

                          {availableClasses.length > 0 ? availableClasses.map(c => (

                            <div key={c.period} onClick={() => togglePeriod(c.period)} className={`period-chip cursor-pointer rounded-xl p-2 text-center text-xs shadow-sm ${currentForm.selectedPeriods.includes(c.period) ? 'active' : 'text-slate-600'}`}>

                              <div className="font-bold mb-1">คาบ {c.period}</div>

                              <div className="text-[10px] truncate">{c.subject}</div>

                            </div>

                          )) : <div className="col-span-4 text-center text-xs text-rose-500 py-2 bg-rose-50 rounded-lg">ไม่มีตารางสอนในวัน{getDayNameFromDate(currentForm.date)}</div>}

                        </div>

                      </div>

                    )}


                    <button onClick={handleAddLeave} className="w-full bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-700 hover:to-indigo-700 text-white font-bold py-3 rounded-2xl shadow-lg shadow-indigo-200 transition-all">

                      + เพิ่มรายชื่อลงคิว

                    </button>

                  </div>

                </div>


                {leaveRequests.length > 0 && (

                  <div className="glass-card rounded-3xl p-5 border-l-4 border-amber-400">

                    <h3 className="text-sm font-bold text-amber-600 mb-3 uppercase">คิวรอประมวลผล ({leaveRequests.length})</h3>

                    <div className="space-y-2">

                      {leaveRequests.map(l => (

                        <div key={l.id} className="flex justify-between items-center text-sm bg-white p-3 rounded-xl border border-slate-100 shadow-sm">

                          <span className="font-medium text-slate-700">

                            {l.teacherName} <span className="text-[10px] bg-slate-100 px-2 py-0.5 rounded text-slate-500 mx-1">{l.reason}</span>

                            <div className="text-xs text-slate-400 mt-1">คาบที่: {l.selectedPeriods.join(', ')}</div>

                          </span>

                          <button onClick={() => setLeaveRequests(leaveRequests.filter(x => x.id !== l.id))} className="text-rose-500 hover:text-rose-700 font-bold bg-rose-50 px-2 py-1 rounded">ลบ</button>

                        </div>

                      ))}

                    </div>

                    <button onClick={calculateSubstitutes} className="w-full mt-4 bg-emerald-500 hover:bg-emerald-600 text-white py-3 rounded-xl text-sm font-bold shadow-md shadow-emerald-200 transition-all">

                      ✨ เริ่มจัดครูสอนแทนอัตโนมัติ

                    </button>

                  </div>

                )}

              </div>


              <div className="lg:col-span-8 space-y-8">

                {assignments.length > 0 ? (

                  <div className="glass-card rounded-3xl overflow-hidden shadow-xl animate-in fade-in duration-500">

                    

                    {leaveRequests.map(leaveReq => {

                      const reqAssignments = assignments.filter(a => a.leaveId === leaveReq.id);

                      if (reqAssignments.length === 0) return null;

                      

                      return (

                        <div key={leaveReq.id} className="mb-6">

                          <div className="bg-gradient-to-r from-rose-50 to-white px-6 py-4 border-b border-rose-100 flex items-center justify-between">

                            <div className="flex items-center">

                              <div className="w-10 h-10 rounded-full bg-rose-100 text-rose-600 flex items-center justify-center font-bold text-xl mr-3">👤</div>

                              <div>

                                <div className="text-sm text-rose-500 font-bold">ครูที่ลา:</div>

                                <div className="text-lg font-bold text-rose-700">{leaveReq.teacherName} <span className="text-sm font-normal bg-white border border-rose-200 px-2 py-0.5 rounded-full ml-2 text-rose-500">{leaveReq.reason}</span></div>

                              </div>

                            </div>
